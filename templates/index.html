Python Log Parser Client Integration (requests library) 
import requests

BASE_URL = "http://127.0.0.1:5000"
API_KEY = "dev-secret-key-12345"

records = [
    {
        "id": "EVT-001",
        "timestamp": "2026-10-29 08:30:00",
        "type": "Database",
        "message": "Database connection refused on port 5432",
        "severity": "Critical"
    }
]

response = requests.post(
    f"{BASE_URL}/api/logs",
    json={"errors": records},
    headers={"X-API-Key": API_KEY, "Content-Type": "application/json"}
)

print(response.status_code, response.json())

      Score = max(0, min(100, 100 - (Critical×12 + Error×4 + Warning×1.5) × DensityFactor - AnomalyPenalty - ConcentrationPenalty))
